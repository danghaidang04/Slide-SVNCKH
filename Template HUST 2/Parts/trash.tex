\begin{frame}{State Space for Multi-Agent RWG (DEC-POSMDP)}
\vspace{-10mm}
\small

We write each global state as
\[
s = \big\langle
  D_{\mathrm{pool}},
  D_{\mathrm{draft}},
  D_{\mathrm{outline}},
  M_{\mathrm{cache}},
  H_{\mathrm{int}}
\big\rangle,
\]
and the state space as
\[
\mathcal{S}
= \mathcal{D}_{\mathrm{pool}}
  \times \mathcal{D}_{\mathrm{draft}}
  \times \mathcal{D}_{\mathrm{outline}}
  \times \mathcal{M}_{\mathrm{cache}}
  \times \mathcal{H}_{\mathrm{int}}.
\]

\begin{itemize}\itemsep2pt
  \item $D_{\mathrm{pool}} \in \mathcal{D}_{\mathrm{pool}}$:
        document pool ( citation papers, keywords, abstracts).
  \item $D_{\mathrm{draft}} \in \mathcal{D}_{\mathrm{draft}}$:
        current RW draft (text + accepted / rejected sentences, scores).
  \item $D_{\mathrm{outline}} \in \mathcal{D}_{\mathrm{outline}}$:
        RW outline (sections, subsections, topic slots, coverage vector).
  \item $M_{\mathrm{cache}} \in \mathcal{M}_{\mathrm{cache}}$:
        shared memory / cache (topic vectors, salience scores, input).
  \item $H_{\mathrm{int}} \in \mathcal{H}_{\mathrm{int}}$:
        interaction history (past macro-actions and intermediate rewards).
\end{itemize}

\end{frame}

\begin{frame}{Action Space for Multi-Agent RWG (DEC-POSMDP)}
\vspace{-8mm}
\small

\[
\mathcal{A}_i =
\{ a^{\mathrm{plan}},\;
   a^{\mathrm{search}},\;
   a^{\mathrm{read}},\;
   a^{\mathrm{write}},\;
   a^{\mathrm{review}} \}.
\]

\begin{columns}[T,totalwidth=\textwidth]

% ===== LEFT COLUMN =====
\begin{column}{0.48\textwidth}
\textbf{Planning} ($a^{\mathrm{plan}}$)
\begin{itemize}\itemsep1pt
  \item Update / refine outline.
  \item Allocate topic slots.
\end{itemize}

\textbf{Search / Retrieval} ($a^{\mathrm{search}}$)
\begin{itemize}\itemsep1pt
  \item Retrieve papers or citation sentences.
  \item Topic- or graph-guided retrieval.
\end{itemize}

\textbf{Reading} ($a^{\mathrm{read}}$)
\begin{itemize}\itemsep1pt
  \item Read selected papers/spans.
  \item Produce summaries, key points.
\end{itemize}
\end{column}

% ===== RIGHT COLUMN =====
\begin{column}{0.48\textwidth}

\textbf{Writing} ($a^{\mathrm{write}}$)
\begin{itemize}\itemsep1pt
  \item Generate RW sentences/paragraphs.
  \item Fill outline slots using Reader outputs.
\end{itemize}

\textbf{Review / Critique} ($a^{\mathrm{review}}$)
\begin{itemize}\itemsep1pt
  \item Check coherence, redundancy, factuality.
  \item Accept/reject/edit generated text.
\end{itemize}

\vspace{2mm}
\textbf{Joint Action Space}
\[
\mathcal{A} = \mathcal{A}_1 \times \dots \times \mathcal{A}_N.
\]

\end{column}

\end{columns}

\vfill
\scriptsize
\hrulefill\\[0.3em]
\textit{Macro-actions operate at high semantic level, enabling long-horizon RW generation.}
\end{frame}


\begin{frame}{Observation Vector}
\vspace{-15mm}
\begin{columns}[T] % The "T" option aligns the columns at the top

\begin{column}{0.5\textwidth}
For each action \( a \in \mathcal{A} \), the number of agents \( n_a(t) \) performing that action at time \( t \) is:

\[
n_a(t) = \sum_{j=1}^{N} \mathbf{1}\{a_j(t) = a\}
\]


The total number of agents \( N \) at time \( t \) is:

\[
\sum_{a \in \mathcal{A}} n_a(t) = \sum_{a \in \mathcal{A}} n_a(t)
\]
\end{column}

\begin{column}{0.5\textwidth}
Apply the function \( F \) to the ratio of the frequency of each action:

\[
s_a(t) = F\left( \frac{n_a(t)}{\sum_{b \in \mathcal{A}} n_b(t)} \right)
\]


To ensure that \( s_a(t) \) stays within the range \([0, 1]\), we use the following sigmoid function \( F \):

\[
F(x) = \frac{1}{1 + e^{-k(x - 0.5)}}
\]
\end{column}

\end{columns}

\end{frame}
